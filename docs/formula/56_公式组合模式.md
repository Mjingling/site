# 公式组合模式

本文档收录所有函数的常见跨函数组合用法，按场景分类，每个组合展示涉及的函数和典型写法。

---

## 一、校验与空值判断

涉及函数：`ISNULL` `NOT_NULL` `IF` `TRIM` `AND` `OR` `NOT_CONTAINS`

### 必填项校验
```js
// 单控件为空时报错
ISNULL(控件)

// A、B 同时为空时报错
ISNULL(A) AND ISNULL(B)

// 等价写法：至少一个不为空才放行
NOT_NULL(A) OR NOT_NULL(B)

// 条件必填
IF(合同类型 == "商务合同" AND ISNULL(附件), true, false)
```

### 空值默认值
```js
// 为空时显示默认文本
IF(ISNULL(备注), "无备注", 备注)

// 为空时默认为 0
IF(ISNULL(折扣率), 0, 折扣率)

// 空值兜底后参与计算
单价 * (1 - IF(ISNULL(折扣率), 0, 折扣率))

// NOT_NULL 与提取函数组合
IF(NOT_NULL(负责人), TEXTUSER(负责人, 'name'), "未指定")
```

### 空格处理
```js
// 先去除首尾空格再判断，纯空格也算空
ISNULL(TRIM(字段))

// 清理后比对
IF(TRIM(备注) == "", "请填写备注", 备注)
```

### 排除判断
```js
// 不包含某关键词
NOT_CONTAINS(商品名称, "测试")

// 多个字段都不为空
NOT_NULL(联系人) AND NOT_NULL(电话)
```

## 二、条件分支

涉及函数：`IF` `CASE` `AND` `OR` `CONTAINS` `NOT_CONTAINS` `STARTSWITH` `MATCH` `TEXTENUM`

### CASE 替代多层 IF
```js
// ❌ 三层 IF 嵌套
IF(成绩 < 60, "不及格", IF(成绩 < 80, "中", IF(成绩 < 90, "良", "优")))

// ✅ CASE 平铺（最后一个参数为默认值）
CASE(成绩 < 60, "不及格", 成绩 < 80, "中", 成绩 < 90, "良", "优")
```

### CONTAINS 关键词分类
```js
IF(CONTAINS(商品名称, "手机"), "数码", IF(CONTAINS(商品名称, "鞋"), "服装", "其他"))

// OR 组合多关键词
IF(CONTAINS(地址, "北京") OR CONTAINS(地址, "上海") OR CONTAINS(地址, "深圳"), "一线城市", "其他城市")
```

### STARTSWITH / MATCH 精确匹配
```js
// 前缀判断
IF(STARTSWITH(工号, "EMP"), "正式员工", "外包员工")
IF(STARTSWITH(手机号, "1"), "有效号码", "无效号码")

// 正则匹配：手机号格式
MATCH(手机号, "^1[3-9]\\d{9}$")
```

### TEXTENUM 枚举值条件
```js
IF(TEXTENUM(状态, 'value') == '2', "进行中", "已完成")
IF(TEXTENUM(优先级, 'value') == '3' OR TEXTENUM(优先级, 'value') == '2', "加急处理", "正常处理")

// TEXTENUM 取 name 与文本比较
IF(TEXTENUM(类型, 'name') == "商务合同", "需法务审批", "常规审批")
```

### AND / OR 组合条件
```js
// 区间判断
(金额 > 1000 AND 金额 <= 5000)

// 任一条件满足
(是否会员 OR 消费金额 > 500)

// 包含A但不包含B
CONTAINS(备注, "加急") AND NOT_CONTAINS(备注, "已处理")
```

## 三、文本处理

涉及函数：`CONCATENATE` `LEFT` `RIGHT` `FIND` `MID` `LEN` `TEXT` `VALUE` `TRIM` `MATCH`

### CONCATENATE 拼接
```js
// 基础拼接
CONCATENATE(省, 市, 区, 详细地址)

// 带分隔符
CONCATENATE(省, "-", 市, "-", 区, "-", 详细地址)

// 对象/数组控件需先提取
CONCATENATE("审批人：", TEXTUSER(审批人, 'name'))
CONCATENATE("部门：", TEXTDEPT(所属部门, 'name'))

// 条件拼接
CONCATENATE(商品名称, IF(是否促销, " [促销]", ""), IF(是否新品, " [新品]", ""))
```

### LEFT / RIGHT 截取
```js
// 前 6 位（地区码）
LEFT(身份证号, 6)

// 后 4 位
RIGHT(身份证号, 4)

// 脱敏：138****1234
LEFT(手机号, 3) + "****" + RIGHT(手机号, 4)

// 补齐位数：001, 002, ...
RIGHT("000" + 流水号, 3)
```

### FIND 提取中间文本
```js
// 返回："100"
FIND("价格:100元", "价格:", "元")

// 提取括号内内容
FIND(全称, "(", ")")

// 取后面的内容
FIND("订单ID:ORD-001", "订单ID:", "")
```

### TEXT + LEN 校验位数
```js
// 数字控件值需 TEXT 转文本后 LEN 才有意义
// 校验 11 位手机号
LEN(TEXT(数字输入框)) == 11

// 校验 18 位身份证
LEN(TEXT(身份证号)) == 18
```

### VALUE 文本转数字
```js
// 转数字后相加
VALUE("100") + VALUE("200")

// 先去空格再转
VALUE(TRIM(" 123 "))

// 枚举值转数字参与计算
VALUE(TEXTENUM(数量, 'value')) * 单价
```

### TRIM 去空格
```js
// 去除首尾空格
TRIM(备注)

// 去空格后转数字
VALUE(TRIM(金额文本))

// 空格也算空
ISNULL(TRIM(字段))

// 去空格后取姓
LEFT(TRIM(姓名), 1)
```

### MATCH 正则校验
```js
// 手机号格式
MATCH(手机号, "^1[3-9]\\d{9}$")

// 邮箱格式
MATCH(邮箱, "^\\S+@\\S+\\.\\S+$")

// 身份证格式
MATCH(身份证号, "^\\d{17}[\\dXx]$")
```

## 四、数学计算

涉及函数：`SUM` `AVERAGE` `MAX` `MIN` `COUNT` `COUNTIF` `SUMIF` `ABS` `ROUND` `ROUNDDOWN` `ROUNDUP` `VALUE`

### 四则运算
```js
// 乘法
单价 * 数量

// 带折扣
(单价 * 数量) * (1 - 折扣率)

// 加运费
单价 * 数量 + 运费

// 利润率百分比
(收入 - 支出) / 收入 * 100
```

### SUM / AVERAGE / MAX / MIN 聚合
```js
// 三科总分
SUM(数学, 语文, 英语)

// 三科平均分
AVERAGE(数学, 语文, 英语)

// 最高分
MAX(数学, 语文, 英语)

// 最低分
MIN(数学, 语文, 英语)

// 摘要统计
// 平均分保留 1 位小数
ROUND(AVERAGE(成绩明细.分数), 1)

// 极差
MAX(成绩明细.分数) - MIN(成绩明细.分数)
```

### COUNT / COUNTIF 计数
```js
// 明细行数
COUNT(明细.控件自身)

// 非空参数个数
COUNT(备选1, 备选2, 备选3)

// COUNTIF 条件计数
// 等级为"优"的行数
COUNTIF(明细.控件自身, 明细.质量等级 == "优", 明细.数量)

// 金额超 1000 的行数
COUNTIF(明细.控件自身, 明细.金额 > 1000)
```

### SUMIF 条件求和
```js
// 已完成订单金额
SUMIF(明细.控件自身, 明细.状态 == "已完成", 明细.金额)

// 金额超 100 的总和
SUMIF(明细.控件自身, 明细.金额 > 100, 明细.金额)

// 类别含"办公"的金额
SUMIF(明细.控件自身, CONTAINS(明细.类别, "办公"), 明细.金额)
```

### ROUND / ROUNDDOWN / ROUNDUP 取整
```js
// 保留 2 位小数，四舍五入
ROUND(金额, 2)

// 取整到元
ROUND(金额 * 税率, 0)

// 向下取整（去尾）
ROUNDDOWN(金额, 0)

// 向上取整（进一）
ROUNDUP(金额, 0)

// 税费计算：嵌套取整
ROUND(ROUND(含税金额 / (1 + 税率), 2) * 税率, 2)
```

### ABS 绝对值
```js
// 预算偏差绝对值
ABS(实际值 - 预算值)

// 盈亏绝对值
ABS(收入 - 支出)

// 偏差检测
IF(ABS(金额1 - 金额2) > 100, "偏差过大", "正常")
```

## 五、日期时间

涉及函数：`NOW` `TODAY` `DATEDIF` `DAYS` `DATEFORMAT` `ADDDAY` `ADDMONTH` `ADDYEAR` `TIMESTAMP` `YEAR` `MONTH` `DAY` `HOUR` `MINUTE` `SECOND` `WEEKDAY` `QUARTER`

### NOW / TODAY 当前时间
```js
// 当前时间戳
NOW()

// 当天 0 点的日期对象
TODAY()

// 当前日期文本
DATEFORMAT(NOW(), "yyyy-MM-dd")

// 当天日期
DATEFORMAT(TODAY(), "yyyy/MM/dd")
```

### DATEDIF / DAYS 时间差
```js
// 相差天数
DATEDIF(开始日期, 结束日期, 'd')

// 相差月数
DATEDIF(开始日期, 结束日期, 'M')

// 相差小时
DATEDIF(开始日期, 结束日期, 'h')

// 相差分钟
DATEDIF(开始日期, 结束日期, 'm')

// 相差秒
DATEDIF(开始日期, 结束日期, 's')

// 相差天数（与 DATEDIF 'd' 等价）
DAYS(开始日期, 结束日期)

// 请假天数校验：超过 5 天报错
DATEDIF(开始日期, 结束日期, 'd') > 5

// 距截止不到 3 天
DATEDIF(NOW(), 截止日期, 'd') <= 3
```

### YEAR / MONTH / DAY / HOUR / MINUTE / SECOND 提取
```js
// 年份
YEAR(日期)

// 月份
MONTH(日期)

// 日期（天）
DAY(日期)

// 小时
HOUR(日期)

// 分钟
MINUTE(日期)

// 秒
SECOND(日期)

// 星期几
WEEKDAY(日期)

// 季度
QUARTER(日期)

// 只能填写当月
IF(MONTH(日期) != MONTH(NOW()) OR YEAR(日期) != YEAR(NOW()), true, false)

// 每月 25 号后不允许提交
DAY(NOW()) >= 25

// 仅工作日可提交
WEEKDAY(日期) >= 2 AND WEEKDAY(日期) <= 6
```

### ADDDAY / ADDMONTH / ADDYEAR 日期加减
```js
// 明天
ADDDAY(日期, 1)

// 昨天
ADDDAY(日期, -1)

// 30 天后
ADDDAY(日期, 30)

// 下个月
ADDMONTH(日期, 1)

// 3 个月前
ADDMONTH(日期, -3)

// 明年
ADDYEAR(日期, 1)

// 去年
ADDYEAR(日期, -1)

// 日期加时间: 日期 + 毫秒数
// 加 30 分钟
日期 + 30 * 60 * 1000
```

### DATEFORMAT 格式化显示
```js
// 2024-03-15
DATEFORMAT(日期, "yyyy-MM-dd")

// 2024/03/15 14:30
DATEFORMAT(日期, "yyyy/MM/dd HH:mm")

// 2024年03月15日
DATEFORMAT(日期, "yyyy年MM月dd日")

// 拼接格式化日期
CONCATENATE("下单日期：", DATEFORMAT(订单日期, "yyyy-MM-dd"))
CONCATENATE(DATEFORMAT(开始日期, "MM/dd"), " ~ ", DATEFORMAT(结束日期, "MM/dd"))
```

### TIMESTAMP 文本转日期
```js
// 文本转时间戳
TIMESTAMP("2024-03-15")

// 带时间
TIMESTAMP("2024/03/15 14:30")

// 本月 1 号
TIMESTAMP(YEAR(NOW()) + "/" + MONTH(NOW()) + "/01")

// 日期比较（必须用时间戳）
主表单.申请日期 < TIMESTAMP("2024-03-15")

// 开始时间在未来
TIMESTAMP(开始文本) > NOW()
```

### 日期字段提取后拼接编号
```js
// 返回：ORD-2024-3-001
CONCATENATE("ORD-", YEAR(订单日期), "-", MONTH(订单日期), "-", 流水号)

// 返回：2024Q2-001
CONCATENATE(YEAR(NOW()), "Q", QUARTER(NOW()), "-", 流水号)
```

## 六、明细表操作

涉及函数：`SUM` `COUNT` `COUNTIF` `SUMIF` `AVERAGE` `MAX` `MIN` `MAP` `ISNULL` `CONTAINS`

### 明细汇总
```js
// 自动汇总明细某列
SUM(费用明细.金额)

// 明细平均分
AVERAGE(成绩明细.分数)

// 明细最高报价
MAX(报价明细.单价)

// 明细最低报价
MIN(报价明细.单价)

// 先乘后汇总
SUM(订单明细.数量 * 订单明细.单价)

// 平均分保留两位
ROUND(AVERAGE(评分明细.分数), 2)
```

### 条件统计
```js
// 通过的数量
COUNTIF(明细.控件自身, 明细.状态 == "通过", 明细.数量)

// 有备注的行数
COUNTIF(明细.控件自身, NOT_NULL(明细.备注))

// 办公用品总金额
SUMIF(明细.控件自身, 明细.类别 == "办公用品", 明细.金额)

SUMIF(明细.控件自身, CONTAINS(明细.名称, "电脑"), 明细.金额)
```

### 明细行数校验
```js
// 明细不能为空
COUNT(明细.控件自身) == 0

// 明细不能超过 10 行
COUNT(明细.控件自身) > 10

// 至少 3 行
COUNT(明细.控件自身) >= 3
```

### MAP 明细逐行校验
```js
// 控件校验（校验控件为 明细.含税金额）
SUM(明细.不含税金额, 明细.税额) != 明细.含税金额

// 提交校验（用 MAP 遍历明细行）
// 任一金额为负则报错
MAP(明细.控件自身, 明细.金额 < 0, true)

// 任一行缺必填项
MAP(明细.控件自身, ISNULL(明细.名称) OR ISNULL(明细.单价), true)

// 任一行超 30 天
MAP(明细.控件自身, DATEDIF(明细.开始日期, 明细.结束日期, 'd') > 30, true)
```

### 主表单与明细交叉校验
```js
// 提交校验：报销金额 ≤ 发票金额
SUM(报销明细.金额) > 发票总金额

// 明细某字段合计后与主表单比较
SUM(预算明细.金额) > 预算总额

// 有不及格项
COUNTIF(明细.控件自身, 明细.分数 < 60) > 0
```

## 七、人员/部门/枚举数据提取

涉及函数：`TEXTUSER` `TEXTDEPT` `TEXTENUM` `GET_PROPERTY` `CONCATENATE` `IF` `UNIQ`

### TEXTUSER 人员信息
```js
// 姓名
TEXTUSER(人员控件, 'name')

// 部门名
TEXTUSER(人员控件, 'department')

// 头像地址
TEXTUSER(人员控件, 'photoUrl')

// 限制特定人员提交
IF(TEXTUSER(主表单.提交人, "name") == "孙悟空", true, false)
IF(TEXTUSER(提交人, "name") == "张三" OR TEXTUSER(提交人, "name") == "李四", true, false)

// 拼接完整人员信息
CONCATENATE(TEXTUSER(负责人, 'name'), "(", TEXTUSER(负责人, 'department'), ")")
```

### TEXTDEPT 部门信息
```js
// 部门名称
TEXTDEPT(部门控件, 'name')

// 部门 ID
TEXTDEPT(部门控件, 'id')

IF(TEXTDEPT(所属部门, 'name') == "财务部", "需财务总监审批", "常规审批")
```

### TEXTENUM 枚举信息
```js
// 显示名称
TEXTENUM(枚举控件, 'name')

// 实际值
TEXTENUM(枚举控件, 'value')

// 优先级系数
IF(TEXTENUM(优先级, 'value') == '3', 1.5, IF(TEXTENUM(优先级, 'value') == '2', 1.2, 1.0))

// 枚举值转数字
VALUE(TEXTENUM(等级, 'value'))
```

### GET_PROPERTY 通用取属性
```js
// 数组第一个元素的 name
GET_PROPERTY(控件, '0.name')

// 人员 oid
GET_PROPERTY(人员, 'oid')
```

### UNIQ 去重
```js
// 明细中出现了几种类型
UNIQ(明细.类型)

// 是否跨部门
COUNT(UNIQ(明细.部门)) > 1
```

## 八、复杂业务示例

### 阶梯价格
```js
// 默认值（最后一个参数）：>= 100 打七折
CASE(数量 < 10, 单价 * 1.0, 数量 < 50, 单价 * 0.9, 数量 < 100, 单价 * 0.8, 单价 * 0.7)
```

### 到期提醒
```js
IF(DATEDIF(NOW(), 截止日期, 'd') <= 7 AND DATEDIF(NOW(), 截止日期, 'd') > 0, "即将到期", IF(DATEDIF(NOW(), 截止日期, 'd') <= 0, "已过期", "正常"))
```

### 加权评分
```js
ROUND((自评分 * 0.3 + 上级评分 * 0.4 + 同事评分 * 0.3) / 3, 1)
```

### 脱敏显示
```js
IF(角色 == "管理员", 完整手机号, LEFT(完整手机号, 3) + "****" + RIGHT(完整手机号, 4))
```

### 预算控制
```js
// 提交校验：累计支出 + 本次申请 > 年度预算 时报错
SUMIF(历史明细.控件自身, 历史明细.年份 == YEAR(NOW()), 历史明细.金额) + 本次金额 > 年度预算
```

### 自动编号
```js
CONCATENATE(TEXTENUM(类型, 'value'), "-", DATEFORMAT(NOW(), "yyyyMMdd"), "-", RIGHT("0000" + 流水号, 4))
```
